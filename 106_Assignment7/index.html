File: src/App.jsx

import { BrowserRouter, Routes, Route } from "react-router-dom";
import { useState } from "react";
import Input from "./components/Input.jsx";
import Display from "./components/Display.jsx";

function App() {
  const [name, setName] = useState("");
  const [city, setCity] = useState("");

  return (
    <BrowserRouter>
      <Routes>
        <Route
          path="/"
          element={
            <Input
              setName={setName}
              setCity={setCity}
            />
          }
        />

        <Route
          path="/display"
          element={
            <Display
              name={name}
              city={city}
            />
          }
        />
      </Routes>
    </BrowserRouter>
  );
}

export default App;

File: src/components/Input.jsx

import { useNavigate } from "react-router-dom";
import "../App.css";

function Input({ setName, setCity }) {
  const navigate = useNavigate();

  return (
    <div className="container">
      <div className="card">
        <h1>Enter Your Details</h1>

        <input
          type="text"
          placeholder="Enter Name"
          onChange={(e) => setName(e.target.value)}
        />

        <input
          type="text"
          placeholder="Enter City"
          onChange={(e) => setCity(e.target.value)}
        />

        <button onClick={() => navigate("/display")}>
          Next
        </button>
      </div>
    </div>
  );
}

export default Input;

File: src/components/Display.jsx

import "../App.css";

function Display({ name, city }) {
  return (
    <div className="container">
      <div className="card">
        <h1>Your Details</h1>

        <h2>
          Name: <span>{name || "Not entered"}</span>
        </h2>

        <h2>
          City: <span>{city || "Not entered"}</span>
        </h2>
      </div>
    </div>
  );
}

export default Display;
